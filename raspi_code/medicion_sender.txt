import requests
from datetime import datetime

class MedicionSender:
    # Constructor que recibe la IP del servidor (PC) y forma la URL del endpoint
    def __init__(self, ip_pc):
        self.url = f"http://{ip_pc}:8000/api/mediciones/"

    # Envía los datos individuales de medición al servidor
    def send_data(self, temperatura, humedad, calidad_aire, presion):
        data = {
            "fecha_hora": datetime.now().isoformat(),
            "temperatura": temperatura,
            "humedad": humedad,
            "calidad_aire": calidad_aire,
            "presion": presion
        }

        try:
            # Realiza una solicitud POST al servidor con los datos en formato JSON
            response = requests.post(self.url, json=data)
            if response.status_code == 200:
                print("Datos enviados correctamente.")
            else:
                print(f"Error al enviar datos: {response.status_code} {response.text}")
        except requests.exceptions.RequestException as e:
            print(f"Error al enviar datos: {e}")

    # Envía un diccionario JSON completo (por ejemplo, con predicciones u otros datos adicionales)
    def send_json(self, data):
        try:
            response = requests.post(self.url, json=data)
            if response.status_code == 200:
                print("Datos enviados correctamente con predicción.")
            else:
                print(f"Error al enviar datos: {response.status_code} {response.text}")
        except requests.exceptions.RequestException as e:
            print(f"Error al enviar datos: {e}")
